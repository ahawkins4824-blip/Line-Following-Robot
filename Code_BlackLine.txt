const uint8_t LED_PIN  = PIN_PD7;
const uint8_t MOTOR_LEFT = PIN_PA3;
const uint8_t MOTOR_RIGHT = PIN_PA2;
const uint8_t SENSOR_1 = PIN_PD1;
const uint8_t SENSOR_2 = PIN_PD2;
const uint8_t SENSOR_3 = PIN_PD3;
const uint8_t SENSOR_4 = PIN_PD4;
const uint8_t SENSOR_5 = PIN_PD5;

void setup() {
    pinMode(LED_PIN, OUTPUT);  //LEDS AS OUTPUTS
    pinMode(MOTOR_LEFT, OUTPUT); //MOTORS AS OUTPUTS
    pinMode(MOTOR_RIGHT, OUTPUT);
    digitalWrite(LED_PIN, LOW); //LED INITAL OFF
    analogWrite(MOTOR_RIGHT, 0); //MOTOR INITAL OFF
    analogWrite(MOTOR_LEFT, 0); //MOTOR INITAL OFF
    Serial.begin(115200);
    uint8_t i = 0;   
    while (i < 2) {
        digitalWrite(LED_PIN, HIGH);
        delay(500);
        digitalWrite(LED_PIN, LOW);
        delay(500);
        i++;
    }
}

int sensor[5];
int sensorcal[5];
uint16_t sensor_max = 0;
uint8_t dom_sensor = 3;

bool start = true;
float kp = 70;
float kd = 0;
float correction = 0;
int time = 1;
float err = 0;
float prev_err = 0;
float integral = 0;
int leftSpeed = 0;
int rightSpeed = 0;
float derivative = 0;
float position = 0;
float lastDirection = 0;
float total = 0;

int baseSpeed = 80;
int maxSpeed = 120;
int minSpeed = 0;
float cutoff = 4700;

uint16_t diff = 0;

void read_sensors()
{
    sensor[0] = analogRead(SENSOR_4);
    sensor[1] = analogRead(SENSOR_5);
    sensor[2] = analogRead(SENSOR_1);
    sensor[3] = analogRead(SENSOR_2);
    sensor[4] = analogRead(SENSOR_3);
}

void calibrate()
{
   int white[5] = {969, 974-8, 972, 971, 970-5};
   int black[5] = {469, 469+158, 469, 469, 439+9};

  for (int i = 0; i < 5; i++) {

    sensorcal[i] = map(
        sensor[i],
        black[i],
        white[i],
        0,
        1000
    );

    sensorcal[i] = constrain(sensorcal[i], 0, 1000);
  }
  sensorcal[2] = constrain(sensorcal[2], 0, 1000);
  sensorcal[1] = constrain(sensorcal[1], 0, 1000);
  sensorcal[0] = constrain(sensorcal[0], 0, 1000);
  sensorcal[4] = constrain(sensorcal[4], 0, 1000);
}

void find_max()
{
    total = (sensorcal[0] + sensorcal[1] + sensorcal[2] + sensorcal[3] + sensorcal[4]);
    if(total < cutoff) {
        position =
        (( 2.0 * sensorcal[0] +
        1.0 * sensorcal[1] +
        0.0 * sensorcal[2] +
        -1.0 * sensorcal[3] +
        -2.0 * sensorcal[4])
        / total);
        if (position > 0) {
            lastDirection = 1;
        } else if (position < 0) {
            lastDirection = -1;
        };
    } else {
        position = lastDirection * 2;
    }
}

void controller()
{
    err = position;
    derivative = (err - prev_err);
    prev_err = err;
    correction = kp * err + kd * derivative;

    leftSpeed  = baseSpeed + correction;
    rightSpeed = baseSpeed - correction;

    leftSpeed  = constrain(leftSpeed, minSpeed, maxSpeed);
    rightSpeed = constrain(rightSpeed, minSpeed, maxSpeed);

    analogWrite(MOTOR_LEFT, leftSpeed);
    analogWrite(MOTOR_RIGHT, rightSpeed);                                                          
}

void loop()
{
    read_sensors();
    calibrate();
    find_max();
    controller();
    digitalWrite(LED_PIN, HIGH);
    time++;
    if(time == 1500) {
        Serial.print("Total = ");
        Serial.println(total);
        Serial.print("left = ");
        Serial.print(leftSpeed);
        Serial.print(" right = ");
        Serial.println(rightSpeed);
        Serial.print("correction = ");
        Serial.println(correction);
        Serial.println(position);
        Serial.print(sensorcal[0]);
        Serial.print(" ");
        Serial.print(sensorcal[1]);
        Serial.print(" ");
        Serial.print(sensorcal[2]);
        Serial.print(" ");
        Serial.print(sensorcal[3]);
        Serial.print(" ");
        Serial.println(sensorcal[4]);
        Serial.print(" ");
        time = 0;
    }
}
